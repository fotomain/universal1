Compact product ER model: products, options, variants

Every table sits on a numbered level, and every arrow points down. A table only ever depends on levels above it. The write side has 7 tables instead of Plan 2's 17. Variant dimensions live on the product as options (up to 3, e.g. Color and Memory), and every variant stores its choice as plain text in option1–option3, as in Shopify and Medusa. The fast read model from Plan 2 sits underneath unchanged, so the shop stays as fast.

Generated from the actual shop schema, tested on PostgreSQL 16: tables, columns, primary keys, foreign keys, unique indexes and triggers. The test loaded a variable phone (3 colors × 3 memory sizes) and a simple food product, with stock in two locations. It checked the rebuild, the facets, the "Blue AND 512 GB on the same variant" filter, the per-store filter, stock and rating roll-ups, rejection of unknown category IDs and rejection of duplicate option combinations. Laid out with ELK's layered algorithm with one partition per level, so no table can move above a table it depends on.

Create / drop order

CREATE (top → bottom)

category → option_swatch → location → product → variant → review → stock_level → dirty → listing → variant_facet → page → category_facet

DROP (bottom → top)

category_facet → page → variant_facet → listing → dirty → stock_level → review → variant → product → location → option_swatch → category

The WooCommerce import follows the same order: categories first, then products with their options, then variations, then stock. Every reference then resolves on the first pass, except category.parent_id and product.parent_id (self references), which are filled in a second pass.

Arrays and option values (category_ids, option1–3) are not real foreign keys: a trigger rejects unknown category IDs, and the app checks option values. Read-model tables (listing, variant_facet, page, category_facet) have no foreign keys on purpose. They are rebuilt rather than referenced. Deleting a product queues it in dirty, and rebuild_product() then removes its read-model rows. Products are normally archived (status = 'draft'), not deleted, because Woo orders reference them.

SQL (tested on PostgreSQL 16)

Full schema: write model, read model, triggers, rebuild
-- COMPACT (Shopify-style) product model: products + options + variants, with the same fast read model as Plan 2
CREATE EXTENSION IF NOT EXISTS pg_trgm;
CREATE SCHEMA IF NOT EXISTS shop;

-- ---------- L0 reference ----------
CREATE TABLE shop.category (
  id int GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  woo_id bigint UNIQUE,
  parent_id int REFERENCES shop.category(id),
  slug text NOT NULL UNIQUE, name text NOT NULL, description text NOT NULL DEFAULT '',
  menu_order int NOT NULL DEFAULT 0,
  ancestor_ids int[] NOT NULL DEFAULT '{}',          -- self + parents, maintained by trigger
  image jsonb                                        -- {url, w, h, alt}
);
CREATE TABLE shop.option_swatch (                    -- shared display data for option values ("Black" -> #111)
  option_name text NOT NULL,                         -- 'Color'
  value text NOT NULL,                               -- 'Black'
  hex text, sort int NOT NULL DEFAULT 0,
  PRIMARY KEY (option_name, value)
);
CREATE TABLE shop.location (                         -- warehouses / stores (optional inventory module)
  id smallint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  code text NOT NULL UNIQUE, name text NOT NULL,
  kind text NOT NULL DEFAULT 'warehouse' CHECK (kind IN ('warehouse','store','pickup')),
  sells_online boolean NOT NULL DEFAULT true, is_pickup boolean NOT NULL DEFAULT false,
  priority smallint NOT NULL DEFAULT 0, address jsonb
);

-- ---------- L1 product ----------
CREATE TABLE shop.product (
  id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  woo_id bigint UNIQUE,
  type text NOT NULL DEFAULT 'simple' CHECK (type IN ('simple','variable','grouped','external')),
  status text NOT NULL DEFAULT 'draft' CHECK (status IN ('draft','pending','private','publish')),
  visibility text NOT NULL DEFAULT 'visible' CHECK (visibility IN ('visible','catalog','search','hidden')),
  slug text NOT NULL UNIQUE,
  name text NOT NULL,
  short_description text NOT NULL DEFAULT '',
  description text NOT NULL DEFAULT '',
  vendor text,                                       -- brand
  featured boolean NOT NULL DEFAULT false,
  -- up to 3 variant dimensions, in order: option1 / option2 / option3 of every variant
  options jsonb NOT NULL DEFAULT '[]'                -- [{"name":"Color","values":["Black","White"]},{"name":"Memory","values":["128 GB","256 GB"]}]
    CHECK (jsonb_typeof(options) = 'array' AND jsonb_array_length(options) <= 3),
  specs jsonb NOT NULL DEFAULT '{}',                 -- non-variant facts {"Material":"Aluminium","Origin":"LV"} (Woo non-variation attributes)
  filter_specs text[] NOT NULL DEFAULT '{}',         -- which specs keys are shop filters, e.g. {Material}
  tags text[] NOT NULL DEFAULT '{}',
  category_ids int[] NOT NULL DEFAULT '{}',          -- -> category.id (array: kept valid by trigger)
  images jsonb NOT NULL DEFAULT '[]',                -- [{url, w, h, alt, lqip, woo_id}], [0] = main
  links jsonb NOT NULL DEFAULT '{}',                 -- {"upsell":[ids],"crossSell":[ids],"grouped":[ids]}
  parent_id bigint REFERENCES shop.product(id),      -- child of a grouped product
  external_url text, button_text text,
  tax_class text NOT NULL DEFAULT '',                -- Woo tax class slug ('' = standard)
  shipping_class text NOT NULL DEFAULT '',
  unit text NOT NULL DEFAULT 'pcs',
  sold_individually boolean NOT NULL DEFAULT false,
  reviews_allowed boolean NOT NULL DEFAULT true,
  purchase_note text NOT NULL DEFAULT '',
  menu_order int NOT NULL DEFAULT 0,
  rating_avg numeric(3,2) NOT NULL DEFAULT 0, rating_count int NOT NULL DEFAULT 0,
  total_sales int NOT NULL DEFAULT 0,
  meta jsonb NOT NULL DEFAULT '[]',                  -- Woo meta_data verbatim [{id,key,value}]
  created_at timestamptz NOT NULL DEFAULT now(), published_at timestamptz, updated_at timestamptz NOT NULL DEFAULT now(),
  woo_modified_gmt timestamptz,
  sync_status text NOT NULL DEFAULT 'localOnly' CHECK (sync_status IN ('synced','dirty','conflict','error','localOnly'))
);
CREATE INDEX product_categories ON shop.product USING gin (category_ids);
CREATE INDEX product_tags ON shop.product USING gin (tags);

-- ---------- L2 sellable unit ----------
CREATE TABLE shop.variant (
  id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  product_id bigint NOT NULL REFERENCES shop.product(id) ON DELETE CASCADE,
  woo_id bigint UNIQUE,                              -- NULL for the default variant of a simple product
  is_default boolean NOT NULL DEFAULT false,
  option1 text, option2 text, option3 text,          -- values of product.options[0..2]
  position int NOT NULL DEFAULT 0,
  sku text UNIQUE, gtin text,
  price bigint NOT NULL CHECK (price >= 0),          -- regular price, cents
  sale_price bigint CHECK (sale_price >= 0), sale_from timestamptz, sale_to timestamptz,
  manage_stock boolean NOT NULL DEFAULT true,
  stock_qty int,                                     -- = SUM(stock_level.available) of online locations when inventory is used
  stock_status text NOT NULL DEFAULT 'instock' CHECK (stock_status IN ('instock','outofstock','onbackorder')),
  backorders text NOT NULL DEFAULT 'no' CHECK (backorders IN ('no','notify','yes')),
  low_stock_amount int,
  weight_g int, dims_mm int[],                       -- {length,width,height}; NULL = inherit
  image jsonb,
  is_virtual boolean NOT NULL DEFAULT false, is_downloadable boolean NOT NULL DEFAULT false,
  downloads jsonb NOT NULL DEFAULT '[]',             -- [{id, name, file}]
  description text NOT NULL DEFAULT '',
  meta jsonb NOT NULL DEFAULT '[]',
  updated_at timestamptz NOT NULL DEFAULT now(), woo_modified_gmt timestamptz,
  UNIQUE NULLS NOT DISTINCT (product_id, option1, option2, option3)
);
CREATE INDEX ON shop.variant (product_id);
CREATE UNIQUE INDEX variant_one_default ON shop.variant (product_id) WHERE is_default;

CREATE TABLE shop.review (
  id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  product_id bigint NOT NULL REFERENCES shop.product(id) ON DELETE CASCADE,
  woo_id bigint UNIQUE, status text NOT NULL DEFAULT 'approved' CHECK (status IN ('approved','hold','spam','trash')),
  reviewer text NOT NULL, rating smallint NOT NULL CHECK (rating BETWEEN 0 AND 5), review text NOT NULL DEFAULT '',
  verified boolean NOT NULL DEFAULT false, created_at timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX ON shop.review (product_id, created_at DESC) WHERE status = 'approved';

-- ---------- L3 inventory (optional) ----------
CREATE TABLE shop.stock_level (
  variant_id bigint NOT NULL REFERENCES shop.variant(id) ON DELETE CASCADE,
  location_id smallint NOT NULL REFERENCES shop.location(id),
  on_hand int NOT NULL DEFAULT 0, reserved int NOT NULL DEFAULT 0,
  available int GENERATED ALWAYS AS (on_hand - reserved) STORED,
  reorder_point int, bin text, counted_at timestamptz,
  PRIMARY KEY (variant_id, location_id)
);

-- ---------- read model (same idea as Plan 2, text facet keys instead of term ids) ----------
CREATE TABLE shop.dirty (product_id bigint PRIMARY KEY, queued_at timestamptz NOT NULL DEFAULT now());
CREATE SEQUENCE shop.catalog_version;

CREATE TABLE shop.listing (
  product_id bigint PRIMARY KEY,
  slug text NOT NULL, name text NOT NULL, card jsonb NOT NULL,
  category_ids int[] NOT NULL,                       -- with ancestors
  facet_keys text[] NOT NULL,                        -- 'color:black', 'memory:256-gb', 'brand:nova', 'tag:new', 'material:aluminium'
  price_min bigint NOT NULL, price_max bigint NOT NULL,
  on_sale boolean NOT NULL, in_stock boolean NOT NULL, featured boolean NOT NULL,
  in_catalog boolean NOT NULL, in_search boolean NOT NULL,
  menu_order int NOT NULL, total_sales int NOT NULL, rating_avg numeric(3,2) NOT NULL, published_at timestamptz NOT NULL,
  search tsvector NOT NULL
);
CREATE INDEX listing_cat ON shop.listing USING gin (category_ids);
CREATE INDEX listing_facets ON shop.listing USING gin (facet_keys);
CREATE INDEX listing_fts ON shop.listing USING gin (search);
CREATE INDEX listing_trgm ON shop.listing USING gin (name gin_trgm_ops);
CREATE INDEX listing_price ON shop.listing (price_min, product_id);
CREATE INDEX listing_new ON shop.listing (published_at DESC, product_id DESC);
CREATE INDEX listing_pop ON shop.listing (total_sales DESC, product_id DESC);

CREATE TABLE shop.variant_facet (
  variant_id bigint PRIMARY KEY, product_id bigint NOT NULL,
  facet_keys text[] NOT NULL, price bigint NOT NULL, in_stock boolean NOT NULL,
  location_ids smallint[] NOT NULL DEFAULT '{}'      -- locations where this variant is available (store filter)
);
CREATE INDEX vf_keys ON shop.variant_facet USING gin (facet_keys);
CREATE INDEX vf_product ON shop.variant_facet (product_id);

CREATE TABLE shop.page (product_id bigint PRIMARY KEY, slug text NOT NULL UNIQUE, doc jsonb NOT NULL, etag text NOT NULL);
CREATE TABLE shop.category_facet (category_id int NOT NULL, facet_key text NOT NULL, product_count int NOT NULL,
  price_min bigint, price_max bigint, PRIMARY KEY (category_id, facet_key));

-- ---------- functions ----------
CREATE OR REPLACE FUNCTION shop.slugify(t text) RETURNS text LANGUAGE sql IMMUTABLE AS
$$ SELECT trim(both '-' from lower(regexp_replace(coalesce(t, ''), '[^[:alnum:]]+', '-', 'g'))) $$;

CREATE OR REPLACE FUNCTION shop.category_set_ancestors() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  NEW.ancestor_ids := COALESCE((SELECT ancestor_ids FROM shop.category WHERE id = NEW.parent_id), '{}') || NEW.id;
  RETURN NEW;
END $$;
CREATE TRIGGER category_ancestors BEFORE INSERT OR UPDATE OF parent_id ON shop.category
  FOR EACH ROW EXECUTE FUNCTION shop.category_set_ancestors();

-- arrays cannot carry FKs: reject unknown category ids
CREATE OR REPLACE FUNCTION shop.product_check_categories() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  IF EXISTS (SELECT 1 FROM unnest(NEW.category_ids) c WHERE NOT EXISTS (SELECT 1 FROM shop.category WHERE id = c)) THEN
    RAISE EXCEPTION 'unknown category id in %', NEW.category_ids;
  END IF;
  RETURN NEW;
END $$;
CREATE TRIGGER product_categories_valid BEFORE INSERT OR UPDATE OF category_ids ON shop.product
  FOR EACH ROW EXECUTE FUNCTION shop.product_check_categories();

CREATE OR REPLACE FUNCTION shop.effective_price(v shop.variant) RETURNS bigint LANGUAGE sql STABLE AS $$
  SELECT CASE WHEN v.sale_price IS NOT NULL AND (v.sale_from IS NULL OR v.sale_from <= now()) AND (v.sale_to IS NULL OR v.sale_to > now())
              THEN v.sale_price ELSE v.price END $$;

-- facet keys of one variant: its options + product-level brand / tags / filterable specs
CREATE OR REPLACE FUNCTION shop.variant_keys(p shop.product, v shop.variant) RETURNS text[] LANGUAGE sql STABLE AS $$
  SELECT COALESCE(array_agg(DISTINCT k), '{}') FROM (
    SELECT shop.slugify(p.options->0->>'name') || ':' || shop.slugify(v.option1) k WHERE v.option1 IS NOT NULL
    UNION ALL SELECT shop.slugify(p.options->1->>'name') || ':' || shop.slugify(v.option2) WHERE v.option2 IS NOT NULL
    UNION ALL SELECT shop.slugify(p.options->2->>'name') || ':' || shop.slugify(v.option3) WHERE v.option3 IS NOT NULL
    UNION ALL SELECT 'brand:' || shop.slugify(p.vendor) WHERE p.vendor IS NOT NULL
    UNION ALL SELECT 'tag:' || shop.slugify(t) FROM unnest(p.tags) t
    UNION ALL SELECT shop.slugify(s) || ':' || shop.slugify(p.specs->>s) FROM unnest(p.filter_specs) s WHERE p.specs ? s
  ) x $$;

CREATE OR REPLACE FUNCTION shop.rebuild_product(pid bigint) RETURNS void LANGUAGE plpgsql AS $$
DECLARE p shop.product;
BEGIN
  SELECT * INTO p FROM shop.product WHERE id = pid;
  IF NOT FOUND OR p.status <> 'publish' THEN
    DELETE FROM shop.listing WHERE product_id = pid; DELETE FROM shop.variant_facet WHERE product_id = pid;
    DELETE FROM shop.page WHERE product_id = pid; RETURN;
  END IF;

  DELETE FROM shop.variant_facet WHERE product_id = pid;
  INSERT INTO shop.variant_facet (variant_id, product_id, facet_keys, price, in_stock, location_ids)
  SELECT v.id, pid, shop.variant_keys(p, v), shop.effective_price(v), v.stock_status <> 'outofstock',
         COALESCE((SELECT array_agg(location_id ORDER BY location_id) FROM shop.stock_level s WHERE s.variant_id = v.id AND s.available > 0), '{}')
  FROM shop.variant v WHERE v.product_id = pid;

  IF p.parent_id IS NOT NULL OR p.visibility = 'hidden' THEN
    DELETE FROM shop.listing WHERE product_id = pid;
  ELSE
    INSERT INTO shop.listing AS l (product_id, slug, name, card, category_ids, facet_keys, price_min, price_max, on_sale, in_stock,
                                   featured, in_catalog, in_search, menu_order, total_sales, rating_avg, published_at, search)
    SELECT pid, p.slug, p.name,
      jsonb_build_object('id', pid, 'slug', p.slug, 'name', p.name, 'type', p.type, 'image', p.images->0,
        'priceMin', a.pmin, 'priceMax', a.pmax, 'regularMin', a.rmin, 'onSale', a.on_sale, 'inStock', a.in_stock,
        'rating', p.rating_avg, 'ratingCount', p.rating_count, 'vendor', p.vendor,
        'swatches', (SELECT jsonb_agg(jsonb_build_object('v', o.value, 'hex', sw.hex) ORDER BY sw.sort)
                     FROM jsonb_array_elements_text(p.options->0->'values') o(value)
                     JOIN shop.option_swatch sw ON sw.option_name = p.options->0->>'name' AND sw.value = o.value)),
      COALESCE((SELECT array_agg(DISTINCT x) FROM shop.category c, unnest(c.ancestor_ids) x WHERE c.id = ANY(p.category_ids)), '{}'),
      COALESCE((SELECT array_agg(DISTINCT k) FROM shop.variant_facet vf, unnest(vf.facet_keys) k WHERE vf.product_id = pid AND vf.in_stock), '{}'),
      a.pmin, a.pmax, a.on_sale, a.in_stock, p.featured,
      p.visibility IN ('visible','catalog'), p.visibility IN ('visible','search'),
      p.menu_order, p.total_sales, p.rating_avg, COALESCE(p.published_at, p.created_at),
      setweight(to_tsvector('simple', p.name || ' ' || COALESCE(p.vendor, '')), 'A') ||
      setweight(to_tsvector('simple', COALESCE((SELECT string_agg(sku, ' ') FROM shop.variant WHERE product_id = pid), '')), 'A') ||
      setweight(to_tsvector('simple', p.short_description || ' ' || array_to_string(p.tags, ' ')), 'B')
    FROM (SELECT COALESCE(min(shop.effective_price(v)) FILTER (WHERE v.stock_status <> 'outofstock'), min(shop.effective_price(v)), 0) pmin,
                 COALESCE(max(shop.effective_price(v)) FILTER (WHERE v.stock_status <> 'outofstock'), max(shop.effective_price(v)), 0) pmax,
                 COALESCE(min(v.price), 0) rmin,
                 COALESCE(bool_or(shop.effective_price(v) < v.price), false) on_sale,
                 COALESCE(bool_or(v.stock_status <> 'outofstock'), false) in_stock
          FROM shop.variant v WHERE v.product_id = pid) a
    ON CONFLICT (product_id) DO UPDATE SET slug = EXCLUDED.slug, name = EXCLUDED.name, card = EXCLUDED.card,
      category_ids = EXCLUDED.category_ids, facet_keys = EXCLUDED.facet_keys, price_min = EXCLUDED.price_min, price_max = EXCLUDED.price_max,
      on_sale = EXCLUDED.on_sale, in_stock = EXCLUDED.in_stock, featured = EXCLUDED.featured, in_catalog = EXCLUDED.in_catalog,
      in_search = EXCLUDED.in_search, menu_order = EXCLUDED.menu_order, total_sales = EXCLUDED.total_sales,
      rating_avg = EXCLUDED.rating_avg, published_at = EXCLUDED.published_at, search = EXCLUDED.search;
  END IF;

  INSERT INTO shop.page (product_id, slug, doc, etag)
  SELECT pid, p.slug, d, md5(d::text) FROM (SELECT jsonb_build_object(
    'product', to_jsonb(p) - 'meta' - 'sync_status' - 'woo_modified_gmt',
    'variants', (SELECT COALESCE(jsonb_object_agg(concat_ws('|', v.option1, v.option2, v.option3),
                   jsonb_build_object('id', v.id, 'sku', v.sku, 'price', shop.effective_price(v), 'regular', v.price,
                                      'stock', v.stock_status, 'qty', CASE WHEN v.manage_stock THEN v.stock_qty END, 'image', v.image,
                                      'locations', (SELECT jsonb_object_agg(l.code, s.available) FROM shop.stock_level s JOIN shop.location l ON l.id = s.location_id
                                                    WHERE s.variant_id = v.id AND s.available > 0))), '{}')
                 FROM shop.variant v WHERE v.product_id = pid),
    'categories', (SELECT COALESCE(jsonb_agg(jsonb_build_object('id', c.id, 'slug', c.slug, 'name', c.name)), '[]') FROM shop.category c WHERE c.id = ANY(p.category_ids))
  ) d) x
  ON CONFLICT (product_id) DO UPDATE SET slug = EXCLUDED.slug, doc = EXCLUDED.doc, etag = EXCLUDED.etag;
  PERFORM nextval('shop.catalog_version');
END $$;

CREATE OR REPLACE FUNCTION shop.mark_dirty() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE r jsonb := to_jsonb(COALESCE(NEW, OLD)); pid bigint;
BEGIN
  pid := COALESCE((r->>'product_id')::bigint, (r->>'id')::bigint);
  IF TG_TABLE_NAME = 'stock_level' THEN SELECT product_id INTO pid FROM shop.variant WHERE id = (r->>'variant_id')::bigint; END IF;
  IF pid IS NOT NULL THEN INSERT INTO shop.dirty (product_id) VALUES (pid) ON CONFLICT DO NOTHING; END IF;
  RETURN NULL;
END $$;
CREATE TRIGGER dirty_product AFTER INSERT OR UPDATE OR DELETE ON shop.product FOR EACH ROW EXECUTE FUNCTION shop.mark_dirty();

-- online stock = sum of available at locations that sell online -> variant.stock_qty / stock_status
CREATE OR REPLACE FUNCTION shop.stock_rollup() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE vid bigint := COALESCE(NEW.variant_id, OLD.variant_id); q int;
BEGIN
  SELECT COALESCE(sum(s.available), 0) INTO q FROM shop.stock_level s JOIN shop.location l ON l.id = s.location_id
  WHERE s.variant_id = vid AND l.sells_online;
  UPDATE shop.variant SET stock_qty = q, stock_status = CASE WHEN q > 0 THEN 'instock' WHEN backorders <> 'no' THEN 'onbackorder' ELSE 'outofstock' END
  WHERE id = vid AND manage_stock;
  RETURN NULL;
END $$;
CREATE TRIGGER stock_rollup AFTER INSERT OR UPDATE OR DELETE ON shop.stock_level FOR EACH ROW EXECUTE FUNCTION shop.stock_rollup();

CREATE OR REPLACE FUNCTION shop.review_rollup() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE pid bigint := COALESCE(NEW.product_id, OLD.product_id);
BEGIN
  UPDATE shop.product p SET rating_avg = COALESCE(r.avg, 0), rating_count = r.cnt
  FROM (SELECT round(avg(rating)::numeric, 2) avg, count(*)::int cnt FROM shop.review WHERE product_id = pid AND status = 'approved') r
  WHERE p.id = pid;
  RETURN NULL;
END $$;
CREATE TRIGGER review_rollup AFTER INSERT OR UPDATE OR DELETE ON shop.review FOR EACH ROW EXECUTE FUNCTION shop.review_rollup();

CREATE OR REPLACE FUNCTION shop.process_dirty(max_items int DEFAULT 500) RETURNS int LANGUAGE plpgsql AS $$
DECLARE r record; n int := 0;
BEGIN
  FOR r IN DELETE FROM shop.dirty WHERE product_id IN (SELECT product_id FROM shop.dirty ORDER BY queued_at LIMIT max_items FOR UPDATE SKIP LOCKED)
           RETURNING product_id LOOP
    PERFORM shop.rebuild_product(r.product_id); n := n + 1;
  END LOOP;
  RETURN n;
END $$;

CREATE OR REPLACE FUNCTION shop.refresh_category_facets() RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  DELETE FROM shop.category_facet;
  INSERT INTO shop.category_facet
  SELECT c, k, count(DISTINCT l.product_id)::int, NULL, NULL
  FROM shop.listing l, unnest(l.category_ids || 0) c, unnest(l.facet_keys) k WHERE l.in_catalog AND l.in_stock GROUP BY 1, 2
  UNION ALL
  SELECT c, '', count(*)::int, min(l.price_min), max(l.price_max)
  FROM shop.listing l, unnest(l.category_ids || 0) c WHERE l.in_catalog AND l.in_stock GROUP BY 1;
END $$;
